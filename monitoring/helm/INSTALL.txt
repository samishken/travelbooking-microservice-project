# Prometheus & Grafana Install Commands

## Step 1: Add Helm repos
helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
helm repo update

## Step 2: Create monitoring namespace and Grafana admin Secret
kubectl create namespace monitoring
kubectl -n monitoring create secret generic grafana-admin \
  --from-literal=admin-user=admin \
  --from-literal=admin-password="$(openssl rand -base64 18 | tr -d '/+=')"

## Step 3: Install kube-prometheus-stack
helm install kube-prometheus-stack prometheus-community/kube-prometheus-stack \
  --namespace monitoring \
  --values monitoring/helm/values.yaml \
  --wait

## Step 4: Apply ServiceMonitors
kubectl apply -f monitoring/servicemonitors/ -n travel-booking

## Step 5: Apply Alert Rules
kubectl apply -f monitoring/alertrules/ -n monitoring

## Step 6: Apply Dashboards
kubectl apply -f monitoring/dashboards/ -n monitoring

## Access Grafana locally
kubectl port-forward svc/kube-prometheus-stack-grafana 3000:80 -n monitoring
# Open: http://localhost:3000/grafana
# Username: admin
# Password: kubectl -n monitoring get secret grafana-admin -o jsonpath='{.data.admin-password}' | base64 -d

## Access Prometheus locally
kubectl port-forward svc/kube-prometheus-stack-prometheus 9090:9090 -n monitoring
# Open: http://localhost:9090/prometheus
